const API_JSON_URL = `https://api.github.com/repos/${REPO_OWNER}/${REPO_NAME}/contents/${FILE_PATH}`; async function initApp() { checkInitialCountry(); let fetched = null; // 1. الجلب اللحظي المباشر من GitHub API (يتجاوز كاش raw.githubusercontent تماماً) try { const apiRes = await fetch(API_JSON_URL + '?ref=main&_=' + Date.now(), { headers: { 'Accept': 'application/vnd.github.v3+json' }, cache: 'no-store' }); if (apiRes.ok) { const meta = await apiRes.json(); const decoded = decodeURIComponent(escape(atob(meta.content.replace(/\s/g, '')))); fetched = JSON.parse(decoded); } } catch (e) {} // 2. مسار احتياطي عبر رابط raw if (!fetched) { try { const res = await fetch(CLOUD_JSON_URL + '?t=' + Date.now(), { cache: 'no-store' }); if (res.ok) fetched = await res.json(); } catch (e) {} } if (fetched) { const mergedBanners = { ...DEFAULT_BANNERS }; if (fetched.banners) { ['b0', 'b1', 'b2', 'b3'].forEach(key => { if (fetched.banners[key]) { mergedBanners[key] = { ...DEFAULT_BANNERS[key], ...fetched.banners[key] }; } }); } appData = { categories: (Array.isArray(fetched.categories) && fetched.categories.length > 0) ? fetched.categories : [...DEFAULT_CATEGORIES], banners: mergedBanners, links: { ...DEFAULT_LINKS, ...(fetched.links || {}) }, stores: Array.isArray(fetched.stores) ? fetched.stores : [], coupons: Array.isArray(fetched.coupons) ? fetched.coupons : [] }; } else { const localSaved = localStorage.getItem('silver_custom_coupons'); if (localSaved) { const parsed = JSON.parse(localSaved); appData = { categories: (Array.isArray(parsed.categories) && parsed.categories.length > 0) ? parsed.categories : [...DEFAULT_CATEGORIES], banners: { ...DEFAULT_BANNERS, ...(parsed.banners || {}) }, links: { ...DEFAULT_LINKS, ...(parsed.links || {}) }, stores: parsed.stores || [], coupons: parsed.coupons || [] }; } } refreshAllUI(); }
